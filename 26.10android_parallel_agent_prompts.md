# StudyHabit Android Audit Fixes — Parallel Execution Prompt Pack (Optimized for Smaller / Weaker Coding Agents)

This document provides a **Copy-Paste Orchestrator Prompt** and **4 Disjoint Parallel Worker Prompts** (+ 1 Final Verification Prompt) engineered specifically for weaker or smaller AI coding models.

## Why This Works Reliably on Weaker Models

1. **Strict File Ownership Isolation (Zero Merge Conflicts):** Workstreams 1, 2, 3, and 4 touch **100% disjoint file sets**. No two parallel agents ever edit the same file.
2. **Pre-Locked Cross-File Signatures:** Every shared method name, parameter type, and default value (`AppContainer.triggerDataChangedSync()`, `FolderSyncService.scheduleDebouncedSync()`, `FolderSyncService.syncStatus`, `SessionDao` project queries, `TimerState.totalWorkSeconds`) is pre-defined identically across agents so all 4 parallel branches compile together cleanly.
3. **Explicit Guardrails for Smaller Models:** Forbids refactoring unrelated files, forbids inventing new dependencies, and provides exact Kotlin code blocks to apply.

### Step 2: Add Keystore Corruption Recovery (`core/network/TokenStorage.kt`)
Wrap `EncryptedSharedPreferences.create(...)` in a helper that catches `Exception`, deletes corrupted `"secure_tokens"` preferences via `context.deleteSharedPreferences("secure_tokens")`, and retries once so Android Keystore invalidation never crashes app startup:
```kotlin
package de.schuelken.studyhabit.core.network

import android.content.Context
import android.content.SharedPreferences
import android.util.Log
import androidx.security.crypto.EncryptedSharedPreferences
import androidx.security.crypto.MasterKey

class TokenStorage(context: Context) {
    private val prefs: SharedPreferences = createEncryptedPrefsWithRecovery(context.applicationContext)

    private fun createEncryptedPrefsWithRecovery(appContext: Context): SharedPreferences {
        return try {
            createPrefs(appContext)
        } catch (e: Exception) {
            Log.w("TokenStorage", "EncryptedSharedPreferences corrupted or key invalidated; resetting", e)
            runCatching { appContext.deleteSharedPreferences("secure_tokens") }
            createPrefs(appContext)
        }
    }

    private fun createPrefs(appContext: Context): SharedPreferences {
        val masterKey = MasterKey.Builder(appContext)
            .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
            .build()
        return EncryptedSharedPreferences.create(
            appContext,
            "secure_tokens",
            masterKey,
            EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
            EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
        )
    }

    fun getToken(): String? = prefs.getString("auth_token", null)

    fun saveToken(token: String) {
        prefs.edit().putString("auth_token", token).apply()
    }

    fun getUserId(): String? = prefs.getString("user_id", null)

    fun getUserEmail(): String? = prefs.getString("user_email", null)

    fun saveUser(id: String, email: String) {
        prefs.edit()
            .putString("user_id", id)
            .putString("user_email", email)
            .apply()
    }

    fun clear() {
        prefs.edit().clear().apply()
    }
}
```

    private fun writeJsonSafely(uri: Uri, jsonContent: String) {
        val bytes = jsonContent.toByteArray(Charsets.UTF_8)
        val pfd = runCatching { context.contentResolver.openFileDescriptor(uri, "rw") }.getOrNull()
        if (pfd != null) {
            pfd.use { descriptor ->
                java.io.FileOutputStream(descriptor.fileDescriptor).use { fos ->
                    fos.channel.truncate(0L)
                    fos.write(bytes)
                    fos.flush()
                    runCatching { descriptor.fileDescriptor.sync() }
                }
            }
            return
        }
        (context.contentResolver.openOutputStream(uri, "wt")
            ?: context.contentResolver.openOutputStream(uri))?.use { os ->
            os.write(bytes)
            os.flush()
        } ?: throw IOException("Could not write to $uri")
    }

### Step 7: Add Foreground/Background Auto-Sync Lifecycle Observer in `StudyHabitApplication.kt`
In `StudyHabitApplication.kt`, register a `DefaultLifecycleObserver` on `ProcessLifecycleOwner.get().lifecycle` so the app automatically syncs with the cloud folder and server when entering foreground (`onStart`) or background (`onStop`), and calls `TimerForegroundService.restoreIfNeeded(this)` on startup:
```kotlin
package de.schuelken.studyhabit

import android.app.Application
import androidx.lifecycle.DefaultLifecycleObserver
import androidx.lifecycle.LifecycleOwner
import androidx.lifecycle.ProcessLifecycleOwner
import de.schuelken.studyhabit.core.di.AppContainer
import de.schuelken.studyhabit.core.service.NotificationHelper
import de.schuelken.studyhabit.core.service.TimerForegroundService
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.launch

class StudyHabitApplication : Application() {
    lateinit var container: AppContainer
        private set

    private val appScope = CoroutineScope(Dispatchers.IO + SupervisorJob())

    override fun onCreate() {
        super.onCreate()
        container = AppContainer(this)
        NotificationHelper.createNotificationChannels(this)
        TimerForegroundService.restoreIfNeeded(this)

        appScope.launch {
            container.database.sessionDao().cleanupOrphanedSessions(
                System.currentTimeMillis() - (2 * 3600_000L)
            )
            container.streakManager.evaluateCurrentStreak()
        }

        ProcessLifecycleOwner.get().lifecycle.addObserver(object : DefaultLifecycleObserver {
            override fun onStart(owner: LifecycleOwner) {
                appScope.launch {
                    if (container.preferences.folderSyncEnabled.value &&
                        !container.preferences.folderSyncUri.value.isNullOrBlank()
                    ) {
                        container.folderSyncService.syncWithFolder()
                        container.streakManager.evaluateCurrentStreak()
                        container.achievementEvaluator.checkAndUnlock()
                    }
                    if (!container.tokenStorage.getToken().isNullOrBlank() &&
                        container.preferences.serverUrl.value.isNotBlank()
                    ) {
                        container.syncEngine.syncNow()
                    }
                }
            }

            override fun onStop(owner: LifecycleOwner) {
                appScope.launch {
                    if (container.preferences.folderSyncEnabled.value &&
                        !container.preferences.folderSyncUri.value.isNullOrBlank()
                    ) {
                        container.folderSyncService.syncWithFolder()
                    }
                }
            }
        })
    }
}
